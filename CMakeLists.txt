    target_compile_options(naviergrain_core PUBLIC -fsanitize=address,undefined,float-cast-overflow -fno-omit-frame-pointer)
    target_link_options(naviergrain_core PUBLIC -fsanitize=address,undefined,float-cast-overflow)

  set(HLOLLI_CSOUND_INCLUDE_DIR "" CACHE PATH "Generated Csound headers")
  set(HLOLLI_CSOUND_BUILD_DIR "${CMAKE_CURRENT_LIST_DIR}/../csound/build" CACHE PATH "Csound build tree")
  set(_naviergrain_header_target "")
  if(NOT HLOLLI_CSOUND_INCLUDE_DIR AND EXISTS "${HLOLLI_CSOUND_BUILD_DIR}/include/csdl.h")
    set(HLOLLI_CSOUND_INCLUDE_DIR "${HLOLLI_CSOUND_BUILD_DIR}/include")
  endif()
  if(NOT HLOLLI_CSOUND_INCLUDE_DIR AND APPLE AND
     EXISTS "${HLOLLI_CSOUND_BUILD_DIR}/CsoundLib64.framework/Headers/csdl.h")
    set(HLOLLI_CSOUND_INCLUDE_DIR "${HLOLLI_CSOUND_BUILD_DIR}/CsoundLib64.framework/Headers")
  endif()
  if(NOT HLOLLI_CSOUND_INCLUDE_DIR)
    find_package(Csound 7 CONFIG QUIET)
    foreach(candidate IN ITEMS Csound::CsoundLib64 Csound::CsoundLib Csound::Csound CsoundLib64 CsoundLib)
      if(TARGET "${candidate}")
        get_target_property(_includes "${candidate}" INTERFACE_INCLUDE_DIRECTORIES)
        if(_includes AND NOT _includes MATCHES "-NOTFOUND$")
          set(_naviergrain_header_target "${candidate}")
          break()
        endif()
      endif()
    endforeach()
  endif()
  if(NOT HLOLLI_CSOUND_INCLUDE_DIR AND NOT _naviergrain_header_target)
    find_path(_naviergrain_system_include csdl.h HINTS "$ENV{CSOUND_INCLUDE_DIR}" PATH_SUFFIXES csound Headers)
    if(_naviergrain_system_include)
      set(HLOLLI_CSOUND_INCLUDE_DIR "${_naviergrain_system_include}")
    endif()
  endif()
  if(HLOLLI_CSOUND_INCLUDE_DIR)
    foreach(header IN ITEMS csdl.h version.h float-version.h)
      if(NOT EXISTS "${HLOLLI_CSOUND_INCLUDE_DIR}/${header}")
        message(FATAL_ERROR "Missing ${header}: use generated <csound-build>/include headers")
      endif()
    endforeach()
  elseif(NOT _naviergrain_header_target)
    message(FATAL_ERROR "Set HLOLLI_CSOUND_BUILD_DIR or HLOLLI_CSOUND_INCLUDE_DIR to a Csound 7 build")
  endif()
  if(APPLE)
    add_library(naviergrain SHARED "${_naviergrain_source}")
  else()
    add_library(naviergrain MODULE "${_naviergrain_source}")
  endif()
  if(HLOLLI_CSOUND_INCLUDE_DIR)
    target_include_directories(naviergrain SYSTEM PRIVATE "${HLOLLI_CSOUND_INCLUDE_DIR}")
  else()
    target_include_directories(naviergrain SYSTEM PRIVATE "$<TARGET_PROPERTY:${_naviergrain_header_target},INTERFACE_INCLUDE_DIRECTORIES>")
  endif()
  include(GNUInstallDirs)
  set(_naviergrain_plugin_folder "plugins64-7.0")
  if(HLOLLI_CSOUND_INCLUDE_DIR)
    file(STRINGS "${HLOLLI_CSOUND_INCLUDE_DIR}/version.h" _major REGEX "^#define[ \t]+CS_VERSION")
    file(STRINGS "${HLOLLI_CSOUND_INCLUDE_DIR}/version.h" _minor REGEX "^#define[ \t]+CS_SUBVER")
    string(REGEX MATCH "[0-9]+" _major "${_major}")
    string(REGEX MATCH "[0-9]+" _minor "${_minor}")
    if(NOT _major STREQUAL "" AND NOT _minor STREQUAL "")
      set(_naviergrain_plugin_folder "plugins64-${_major}.${_minor}")
    endif()
  endif()
  set(_default_install "${CMAKE_INSTALL_LIBDIR}/csound/${_naviergrain_plugin_folder}")

  endif()

    target_include_directories(naviergrain_host_test SYSTEM PRIVATE
      "$<TARGET_PROPERTY:naviergrain,INCLUDE_DIRECTORIES>")
    target_include_directories(naviergrain_host_test PRIVATE "${CMAKE_CURRENT_LIST_DIR}/include")
    target_link_libraries(naviergrain_host_test PRIVATE "${NAVIERGRAIN_CSOUND_LIBRARY}")
    if(UNIX)
      target_link_libraries(naviergrain_host_test PRIVATE m)
    endif()
    target_compile_features(naviergrain_host_test PRIVATE c_std_11)
    if(NAVIERGRAIN_SANITIZE AND CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
      # Instrument the API host too: do not load an ASan plugin into the
      # uninstrumented command-line Csound process.
      target_compile_options(naviergrain_host_test PRIVATE -fsanitize=address,undefined,float-cast-overflow -fno-omit-frame-pointer)
      target_link_options(naviergrain_host_test PRIVATE -fsanitize=address,undefined,float-cast-overflow)
    endif()
    if(UNIX)
      add_executable(naviergrain_worker_host examples/cpu-worker.c)
      target_include_directories(naviergrain_worker_host SYSTEM PRIVATE
        "$<TARGET_PROPERTY:naviergrain,INCLUDE_DIRECTORIES>")
      target_include_directories(naviergrain_worker_host PRIVATE
        "${CMAKE_CURRENT_LIST_DIR}/include" "${CMAKE_CURRENT_LIST_DIR}/src")
      target_link_libraries(naviergrain_worker_host PRIVATE
        "${NAVIERGRAIN_CSOUND_LIBRARY}" Threads::Threads ${CMAKE_DL_LIBS} m)
      target_compile_features(naviergrain_worker_host PRIVATE c_std_11)
      if(NAVIERGRAIN_SANITIZE AND CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(naviergrain_worker_host PRIVATE -fsanitize=address,undefined,float-cast-overflow -fno-omit-frame-pointer)
        target_link_options(naviergrain_worker_host PRIVATE -fsanitize=address,undefined,float-cast-overflow)
      endif()
      add_dependencies(naviergrain_worker_host naviergrain)
      add_test(NAME naviergrain.worker_host COMMAND naviergrain_worker_host "$<TARGET_FILE:naviergrain>")
      set_tests_properties(naviergrain.worker_host PROPERTIES
        WORKING_DIRECTORY "${CMAKE_CURRENT_LIST_DIR}" TIMEOUT 120)
    endif()
    add_dependencies(naviergrain_host_test naviergrain)
    add_test(NAME naviergrain.host COMMAND naviergrain_host_test "$<TARGET_FILE:naviergrain>")
    set_tests_properties(naviergrain.host PROPERTIES
      WORKING_DIRECTORY "${CMAKE_CURRENT_LIST_DIR}" TIMEOUT 120)
    if(TARGET naviergrain_profile AND NOT NAVIERGRAIN_SANITIZE)
      add_test(NAME naviergrain.profile COMMAND "${Python3_EXECUTABLE}"
        "${CMAKE_CURRENT_LIST_DIR}/tests/test_profile.py"
        --module "$<TARGET_FILE:naviergrain>"
        --profile-module "$<TARGET_FILE:naviergrain_profile>"
        --library "${NAVIERGRAIN_CSOUND_LIBRARY}")
      set_tests_properties(naviergrain.profile PROPERTIES TIMEOUT 240 SKIP_RETURN_CODE 77)
    endif()
  else()
    message(STATUS "Prepared-host test unavailable: set NAVIERGRAIN_CSOUND_LIBRARY to the matching Csound 7 library")
  endif()
